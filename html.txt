    // --- 實作動態題庫引擎 (數字改編、洗牌不重複) ---
    function rand(min, max) { return Math.floor(Math.random() * (max - min + 1)) + min; }

    function genOptions(ansStr, unit="") {
        let ansNum = parseFloat(ansStr);
        let opts = new Set([ansStr + unit]);
        
        while(opts.size < 4) {
            let type = Math.floor(Math.random() * 4);
            let wrongStr = "";
            if (type === 0) {
                wrongStr = (ansNum + rand(1, 5) * 10).toString();
            } else if (type === 1) {
                wrongStr = (ansNum * (Math.random() > 0.5 ? 10 : 0.1)).toString();
            } else if (type === 2) {
                let offset = Math.max(1, Math.floor(ansNum * 0.2));
                let val = ansNum - offset;
                wrongStr = (val > 0 ? val : ansNum + offset).toString();
            } else {
                wrongStr = (ansNum * rand(2, 4)).toString();
            }
            // 避免出現浮點數誤差過長
            if (wrongStr.includes('.')) wrongStr = parseFloat(wrongStr).toFixed(2).replace(/\.00$/, '');
            opts.add(wrongStr + unit);
        }
        return Array.from(opts);
    }

    function genLevel1() {

            q = `有一個內部空間是正方體的水族箱，內部的邊長是${l}公分，容積是幾立方公分？`;
            ans = (l * l * l).toString();
        } else if(type === 2) {
            let l = rand(10, 30), w = rand(5, 20), h = rand(2, 10);
            q = `長方體塑膠盒裡面的長${l}公分、寬${w}公分、高${h}公分，容積是幾立方公分？`;
            ans = (l * w * h).toString();
        } else if(type === 3) {
            let n = rand(10, 50);
            q = `用${n}個1立方公尺的箱子剛好排滿儲藏室，儲藏室容積是幾立方公尺？`;
            ans = n.toString();
        } else if(type === 4) {
            let l = rand(2, 8);
            q = `內部邊長是${l}公尺的正方體水池，容積是多少立方公尺？`;
            ans = (l * l * l).toString();
        } else {
            let l = rand(20, 50), w = rand(15, 30), h = rand(10, 20), d = rand(5, 8);
            q = `有一個內部長${l}公分、寬${w}公分、高${h}公分的水箱，水深是${d}公分。水箱的容積是多少立方公分？`;
            ans = (l * w * h).toString(); // 注意是問容積，不是水量
        }
        return { q, ans, options: genOptions(ans) };
    }

    // 關卡二動態生成器
    function genLevel2() {
        let type = rand(1, 6);
        let q, ans;
        if(type === 1) {
            let l = rand(5, 50);
            q = `${l} 公升 = 幾毫升？`;
            ans = (l * 1000).toString();
        } else if(type === 2) {
            let v = rand(2, 20) * 1000;
            q = `${v} 立方公分等於幾公升？`;
            ans = (v / 1000).toString();
        } else if(type === 3) {
            let d = rand(5, 45);
            q = `水費單上顯示用水量是 ${d} 度，代表用了多少公升的水？`;
            ans = (d * 1000).toString();
        } else if(type === 4) {
            let c = rand(2, 10) + (rand(1,9)/10);
            q = `有一個塑膠桶的容量是 ${c} 公升，容積是多少立方公分？`;
            ans = (c * 1000).toString();
        } else if(type === 5) {
            let t = rand(1, 2);
            let l = rand(15, 25), w = rand(10, 15), h = rand(5, 10);
            q = `一個長${l}公分、寬${w}公分、高${h}公分的「有蓋」長方體鐵盒，厚度是${t}公分，這個鐵盒的容量是幾毫升？`;
            ans = ((l - 2*t) * (w - 2*t) * (h - 2*t)).toString();
        } else {
            let t = rand(1, 3);
            let l = rand(20, 30);
            q = `厚度${t}公分的「無蓋」正方體水槽，外面邊長是${l}公分，它的容量是幾毫升？`;
            ans = ((l - 2*t) * (l - 2*t) * (l - t)).toString();
        }
        return { q, ans, options: genOptions(ans) };
    }

    function genLevel3() {
        let type = rand(1, 5);
        let q, ans;
        if(type === 1) {
            let v1 = rand(200, 400);
            let v2 = v1 + rand(50, 150);
            q = `量杯裝有 ${v1} 毫升的水，放入一顆石頭(完全沉入)後水面上升到 ${v2} 毫升，石頭體積是幾立方公分？`;
            ans = (v2 - v1).toString();
        } else if(type === 2) {
            let l = rand(10, 20), w = rand(10, 15), h = rand(1, 3), n = rand(2, 5);
            q = `長方體容器內部長${l}公分、寬${w}公分，放入${n}塊相同磁鐵(完全沉入)後水面上升${h}公分，一塊磁鐵的體積是幾立方公分？`;
            ans = ((l * w * h) / n).toString();
        } else if(type === 3) {
            let l = rand(4, 12);
            q = `將邊長${l}公分的正方體積木完全沉入「裝滿水」的容器，會溢出幾毫升的水？`;
            ans = (l * l * l).toString();
        } else if(type === 4) {
            let l = rand(10, 30), h1 = rand(5, 10), h2 = h1 + rand(2, 5);
            q = `內部邊長${l}公分的正方體水箱，放入一塊石頭後，水深從${h1}公分變為${h2}公分，石頭體積多少立方公分？`;
            ans = (l * l * (h2 - h1)).toString();
        } else {
            let l = rand(30, 50), w = rand(20, 30), h1 = rand(10, 20), h2 = h1 + rand(1, 4), n = rand(2, 10);
            q = `長方體水缸裡面長${l}公分、寬${w}公分，裝水${h1}公分高。放進${n}個芒果後，水位升到${h2}公分，1個芒果平均體積是？`;
            ans = ((l * w * (h2 - h1)) / n).toString();
        }
        return { q, ans, options: genOptions(ans) };
    }

            else if(level === 2) pool.push(genLevel2());

        }
        return pool;
    }

    // --- 遊戲邏輯 ---
    function shuffleArray(array) {
        for (let i = array.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [array[i], array[j]] = [array[j], array[i]];
        }
        return array;
    }

    function loadQuestion() {
        if (currentQIndex >= currentQuestions.length) {
            currentQuestions = shuffleArray(buildDynamicQuestionPool(selectedLevel));
            currentQIndex = 0;

        const qData = currentQuestions[currentQIndex];

        document.getElementById('q-num-display').innerText = totalAnswered + 1;
        
        const optionsGrid = document.getElementById('options-grid');
        optionsGrid.innerHTML = '';
        
        const shuffledOptions = shuffleArray([...qData.options]);

            const btn = document.createElement('button');
            btn.className = 'option-btn';

        if (isPaused) return;

        totalAnswered++;
        let isCorrect = (selected === correct);
        
        reviewData.unshift({
            qNum: totalAnswered,
            q: currentQuestions[currentQIndex].q,
            userAns: selected,
            correctAns: correct,
            isCorrect: isCorrect
        });

        if (isCorrect) {
            combo++;
            score += 10 * combo; 
            timeRemaining += 5;  
            showFeedback(true);
        } else {
            combo = 0;
            timeRemaining -= 5;  
            showFeedback(false, correct);
        }
        
        updateHUD();
        currentQIndex++;
        
        setTimeout(() => {
            document.getElementById('feedback-modal').style.display = 'none';
            if (timeRemaining > 0) loadQuestion();
        }, 1800);
    }

            diff: diffName,

    function setTool(tool) {
        currentTool = tool;
        const penBtn = document.getElementById('wb-pen-btn');
        const eraserBtn = document.getElementById('wb-eraser-btn');

        if(!penBtn || !eraserBtn) return;

        penBtn.classList.remove('active');
        eraserBtn.classList.remove('active-eraser');

        if (tool === 'pen') {
            penBtn.classList.add('active');
            ctx.globalCompositeOperation = 'source-over';
            ctx.strokeStyle = '#2980B9'; 
            ctx.lineWidth = 4;
        } else if (tool === 'eraser') {
            eraserBtn.classList.add('active-eraser');
            ctx.globalCompositeOperation = 'destination-out';
            ctx.lineWidth = 30; 
        }
        ctx.lineCap = 'round';
        ctx.lineJoin = 'round';
    }

    function clearWhiteboard() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
    }

    function getPos(e) {
        const clientX = e.touches ? e.touches[0].clientX : e.clientX;
        const clientY = e.touches ? e.touches[0].clientY : e.clientY;
        return { x: clientX, y: clientY };
    }

    function isInteractiveElement(target) {
        return target.tagName === 'BUTTON' || target.closest('button') || 
               target.tagName === 'A' || target.closest('a') || 
               target.tagName === 'INPUT';
    }

    document.addEventListener('mousedown', (e) => { 
        if(!isWhiteboardActive || isPaused) return; 
        if(isInteractiveElement(e.target)) return; 
        
        e.preventDefault(); 
        drawing = true; 
        const pos = getPos(e); 
        ctx.beginPath(); 
        ctx.moveTo(pos.x, pos.y); 
    });
    
    document.addEventListener('mousemove', (e) => { 
        if(!drawing || !isWhiteboardActive || isPaused) return; 
        const pos = getPos(e); 
        ctx.lineTo(pos.x, pos.y); 
        ctx.stroke(); 
    });
    
    document.addEventListener('mouseup', () => drawing = false);
    document.addEventListener('mouseleave', () => drawing = false);

        if(!isWhiteboardActive || isPaused) return; 
        if(isInteractiveElement(e.target)) return; 

        e.preventDefault(); 
        drawing = true; 
        const pos = getPos(e); 
        ctx.beginPath(); 
        ctx.moveTo(pos.x, pos.y); 
    }, {passive: false});

        if(!drawing || !isWhiteboardActive || isPaused) return; 
        
        e.preventDefault(); 
        const pos = getPos(e); 
        ctx.lineTo(pos.x, pos.y); 
        ctx.stroke(); 
    }, {passive: false});
    
    document.addEventListener('touchend', () => drawing = false);
